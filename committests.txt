
<?php
// Calculator object
class Calculator {
	//function for adding  2 numbers
    public function add($a, $b, $c) {
        return $a + $b + $c;
    }
	//function for subtracting  2 numbers
    public function subtract($a, $b, $c) {
        return $a - $b - $c;
    }
	 public function multiply($a, $b) {
        return $a * $b;
    }
}

$calculator = new Calculator();
echo $calculator->add(5, 3, 10);      
echo $calculator->subtract(5, 3); 

?>
