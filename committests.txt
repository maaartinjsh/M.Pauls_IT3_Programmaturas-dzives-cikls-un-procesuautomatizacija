
<?php
// Calculator object
class Calculator {
	//function for adding  2 numbers
    public function add($a, $b) {
        return $a + $b;
    }

    public function subtract($a, $b) {
        return $a - $b;
    }
}

$calculator = new Calculator();
echo $calculator->add(5, 3);      
echo $calculator->subtract(5, 3); 

?>
