
<?php
// Calculator object
class Calculator {
	//function for adding 3 numbers
    public function add($a, $b, $c) {
        return $a + $b + $c;
    }
	//function for subtracting 3 numbers
    public function subtract($a, $b, $c) {
        return $a - $b - $c;
    }
    // Function for multiplying 2 numbers
	 public function multiply($a, $b, $c)
        return $a * $b * $c;
    }
}

$calculator = new Calculator();
echo $calculator->add(5, 3, 10);      
echo $calculator->subtract(5, 3, 1);

?>
